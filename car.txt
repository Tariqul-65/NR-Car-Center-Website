USE car;

DROP TABLE IF EXISTS stock_images;
DROP TABLE IF EXISTS stock_cars;

CREATE TABLE stock_cars (
    id INT NOT NULL AUTO_INCREMENT,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    stock_id VARCHAR(32) NOT NULL,
    chassis_no VARCHAR(64) NOT NULL,
    make VARCHAR(120) NOT NULL,
    model VARCHAR(120) NOT NULL,
    grade VARCHAR(60) DEFAULT NULL,
    chassis_code VARCHAR(80) DEFAULT NULL,
    year INT NOT NULL,
    color VARCHAR(60) NOT NULL,
    engine_cc INT DEFAULT NULL,
    fuel VARCHAR(20) DEFAULT NULL,              
    mileage INT DEFAULT NULL,
    auction_point DECIMAL(4,1) DEFAULT NULL,     
    status VARCHAR(16) NOT NULL,                 
    price DECIMAL(20,2) NOT NULL,                
    transmission VARCHAR(20) DEFAULT NULL,       
    seats INT DEFAULT NULL,
    doors INT DEFAULT NULL,
    CONSTRAINT pk_stock_cars PRIMARY KEY (id),
    CONSTRAINT uq_stock_id UNIQUE (stock_id),
    CONSTRAINT uq_chassis_no UNIQUE (chassis_no)
);

CREATE INDEX idx_stock_stock_id ON stock_cars (stock_id);
CREATE INDEX idx_stock_chassis  ON stock_cars (chassis_no);
CREATE INDEX idx_stock_status   ON stock_cars (status);

CREATE TABLE stock_images (
    id INT NOT NULL AUTO_INCREMENT,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
    stock_car_id INT NOT NULL,
    image_path VARCHAR(255) NOT NULL,
    CONSTRAINT pk_stock_images PRIMARY KEY (id),
    CONSTRAINT fk_image_stock_car FOREIGN KEY (stock_car_id) REFERENCES stock_cars(id) ON DELETE CASCADE
);

    -- Fields explicitly declared in DeliveredCar.java

    make VARCHAR(120) NOT NULL,
    model VARCHAR(120) NOT NULL,
    customer_name VARCHAR(120) NOT NULL,
    delivery_date DATE NOT NULL,
    delivery_cost DECIMAL(14,2) NOT NULL,
    image_path VARCHAR(255) DEFAULT NULL,
    
    CONSTRAINT pk_delivered_cars PRIMARY KEY (id)
);
